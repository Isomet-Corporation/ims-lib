542 template <typename Ratio>
543 class distance {
544 double ticks;
545 public:
547 distance(double ticks = 1.0) :
548 ticks(ticks)
549 {}
550
553 template <typename Ratio2>
554 distance(distance<Ratio2> other) {
555 using divided = std::ratio_divide<Ratio2, Ratio>;
556 ticks = other.operator double() * divided::num / divided::den;
557 }
558
560 distance& operator = (double arg) {
561 ticks = arg; return *this;
562 }
563
565 operator double() const {
566 return ticks;
567 }
568 };
569
571 template <typename Ratio1, typename Ratio2>
572 bool operator==(distance<Ratio1> d1, distance<Ratio2> d2) {
573 double normalized1 = d1.count() * Ratio1::num / Ratio1::den;
574 double normalized2 = d2.count() * Ratio2::num / Ratio2::den;
575 return normalized1 == normalized2;
576 }
